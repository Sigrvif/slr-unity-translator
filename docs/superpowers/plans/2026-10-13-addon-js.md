```js
const fs = require("fs");
const path = require("path");

function resolveRuntime(basePath) {
  const cli = path.join(basePath, "backend", "slr_cli.py");
  if (process.platform === "win32") {
    const exe = path.join(basePath, "runtime", "python", "python.exe");
    return fs.existsSync(exe) ? { cmd: exe, baseArgs: [cli] } : null;
  }
  const exe = path.join(basePath, "runtime", "bin", "python3");
  return fs.existsSync(exe) ? { cmd: exe, baseArgs: [cli] } : null;
}

function parseProgressLines(chunk, state) {
  state.buf += chunk.toString("utf-8");
  const out = [];
  let idx;
  while ((idx = state.buf.indexOf("\n")) >= 0) {
    const line = state.buf.slice(0, idx).trim();
    state.buf = state.buf.slice(idx + 1);
    if (!line) continue;
    try { const o = JSON.parse(line); if (o && o.type === "progress") out.push(o); } catch (e) {}
  }
  return out;
}

let _jobCounter = 0;
async function runBackend(job, opts) {
  const spawnFn = opts.spawnFn || require("child_process").spawn;
  const python = opts.python || resolveRuntime(opts.basePath || ".");
  if (!python) throw new Error("Unity backend runtime not found (payload missing for this OS)");
  fs.mkdirSync(opts.jobsDir, { recursive: true });
  const id = `job-${Date.now()}-${++_jobCounter}`;
  const manifest = path.join(opts.jobsDir, id + ".json");
  const result = path.join(opts.jobsDir, id + ".result.json");
  fs.writeFileSync(manifest, JSON.stringify(job), "utf-8");
  return await new Promise((resolve, reject) => {
    const child = spawnFn(python.cmd, [...python.baseArgs, "--manifest", manifest, "--result", result],
      { windowsHide: true, cwd: opts.jobsDir });
    const state = { buf: "" };
    child.stdout.on("data", (d) => {
      for (const p of parseProgressLines(d, state)) opts.onProgress && opts.onProgress(p);
    });
    child.stderr.on("data", () => {});   // backend logs its own stderr into backend.log
    child.on("error", reject);
    child.on("close", () => {
      try {
        const payload = JSON.parse(fs.readFileSync(result, "utf-8"));
        resolve(payload);
      } catch (e) {
        reject(new Error("backend produced no result file; see " + path.join(opts.jobsDir, "backend.log")));
      }
    });
  });
}
module.exports = { runBackend, parseProgressLines, resolveRuntime };
```

**Interfaces:**
- Consumes: scan payload per `docs/CONTRACT.md`.
- Produces:
  - `buildProject(scan, meta)` → SLR project object. `meta = {gameDir, gameTitle, cache: {cacheID, cachePath}}`. Shape:
    ```js
    { gameEngine: "unity", parser: "unity", gameTitle, loc: gameDir,
      projectId: cache.cacheID, cache,
      files: { [fileKey]: { path, dirname, basename, extension,
                            data: [[jp, ""]...], context: [[ctx...]...],
                            tags: [["gray"]...], parameters: [[]...] } },
      options: { unity: { contract: 1, gameDir, sceneIndex,
                          stats, fontsFound, texturesFound, textAssetsFound,
                          monoReport, suspect } } }
    ```
  - `buildSceneIndex(files)` → `{ order: [fileKey...], files: { [key]: { kind: "scene"|"table", label, subgroups: [{ name, rows: [{row, leaf}] }] } } }`
    - `order`: keys whose basename matches `level\d+` first (numeric order), then the rest alphabetically.
    - `kind`: `"scene"` for levelN, `"table"` otherwise.
    - `label`: basename (e.g. `level0`).
    - `subgroups`: for scene files, group rows by the first `/`-segment of their FIRST context string (fallback `"_"`); rows in build order; `leaf` = last segment. Table files get one subgroup `"table"`.
  - `sortKeyForFile(fileKey)` → `[0, N]` for levelN else `[1, key]` (exported for tests).
  - Row building rules: per file, dedupe rows by text (first occurrence wins); occurrence order = (path_id, offset, key-string); `context[row]` = unique occurrence context strings in order (cap 20, join nothing — array form).

```js
const path = require("path");

function sortKeyForFile(fileKey) {
  const base = path.basename(fileKey);
  const m = /^level(\d+)$/.exec(base);
  return m ? [0, parseInt(m[1], 10), fileKey] : [1, 0, fileKey];
}

function _rowOrder(o) { return [o.path_id || 0, o.offset || 0, o.key || ""]; }

function buildSceneIndex(files) {
  const order = Object.keys(files).sort((a, b) => {
    const ka = sortKeyForFile(a), kb = sortKeyForFile(b);
    return ka[0] - kb[0] || ka[1] - kb[1] || a.localeCompare(b);
  });
  const out = { order, files: {} };
  for (const key of order) {
    const base = path.basename(key);
    const kind = /^level\d+$/.test(base) ? "scene" : "table";
    const groups = new Map();
    const rows = files[key].context || [];
    rows.forEach((ctxs, row) => {
      const first = (ctxs && ctxs[0]) || "";
      const segs = String(first).split("/");
      const name = kind === "scene" ? (segs[0] || "_") : "table";
      const leaf = segs[segs.length - 1] || name;
      if (!groups.has(name)) groups.set(name, { name, rows: [] });
      groups.get(name).rows.push({ row, leaf });
    });
    out.files[key] = { kind, label: base, subgroups: [...groups.values()] };
  }
  return out;
}

function buildProject(scan, meta) {
  const byFile = new Map();
  for (const [text, occs] of Object.entries(scan.index || {})) {
    const sorted = [...occs].sort((a, b) => _rowOrder(a) < _rowOrder(b) ? -1 : _rowOrder(a) > _rowOrder(b) ? 1 : 0);
    for (const o of sorted) {
      if (!byFile.has(o.file)) byFile.set(o.file, new Map());
      const m = byFile.get(o.file);
      if (!m.has(text)) m.set(text, { path_id: o.path_id, contexts: [] });
      const ctx = _contextFor(o);
      if (ctx && !m.get(text).contexts.includes(ctx) && m.get(text).contexts.length < 20)
        m.get(text).contexts.push(ctx);
    }
  }
  const files = {};
  for (const [fileKey, texts] of byFile) {
    const base = path.basename(fileKey);
    const data = [], context = [], tags = [], parameters = [];
    for (const [text, info] of texts) {
      data.push([text, ""]);
      context.push(info.contexts.length ? info.contexts : [base]);
      tags.push(["gray"]);
      parameters.push([]);
    }
    files[fileKey] = { path: fileKey, dirname: path.dirname(fileKey), basename: base,
                       extension: path.extname(base).slice(1), data, context, tags, parameters };
  }
  const sceneIndex = buildSceneIndex(files);
  return {
    gameEngine: "unity", parser: "unity", gameTitle: meta.gameTitle,
    loc: meta.gameDir, projectId: meta.cache.cacheID, cache: meta.cache,
    files,
    options: { unity: { contract: 1, gameDir: meta.gameDir, sceneIndex,
                        stats: scan.stats || {}, fontsFound: scan.fonts || [],
                        texturesFound: (scan.textures || []).length,
                        textAssetsFound: scan.text_assets || [],
                        monoReport: scan.mono || [], suspect: scan.suspect || [] } },
  };
}

function _contextFor(o) {
  if (o.source === "metadata") return `literal #${o.path_id}`;
  if (o.source === "textfile" || o.source === "textasset") return o.key || "";
  return "";   // raw/typetree occurrence contexts are already in scan strings[] rows
}
```

function detectUnity(fsh, gamePath) {
  const base = fsh.isFile(gamePath) ? fsh.dirname(gamePath) : gamePath;
  const candidates = [base, ...fsh.listDir(base).map(d => fsh.join(base, d))];
  for (const dir of candidates) {
    if (!fsh.isDir(dir)) continue;
    if (fsh.isFile(fsh.join(dir, "UnityPlayer.dll")) || fsh.isFile(fsh.join(dir, "GameAssembly.dll"))) return "unity";
    for (const sub of fsh.listDir(dir)) {
